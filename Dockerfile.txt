
FROM node:22-alpine AS build

WORKDIR /app

COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

FROM node:22-alpine

WORKDIR /app

COPY package*.json ./
RUN npm ci --omit=dev

COPY --from=build /app/src ./src
COPY --from=build /app/dist ./dist
COPY --from=build /app/.env ./.env

ENV NODE_ENV=production

EXPOSE 5000

CMD ["npm", "run", "backend"]
